\section{Research Profile}

I develop robot-learning methods for physical interaction, spanning force-aware
human demonstrations, play-driven dexterity, and interactive
navigation. My work combines learning and planning with custom sensing and
real-robot system validation.

\section{Selected Research \& Systems}

\entry{Dexterity from Supervised Play}{— arXiv coming soon}{}{
  \textit{Learn dexterity from minutes of goal-agnostic human play.}
}

\entry{\href{https://forceband-emg.github.io/}{ForceBand}}{— CoRL 2026}{}{
  \textit{Bring force into human data with EMG, for forceful manipulation.}
}

\entry{\href{https://www.cs.umd.edu/\string~edessale/feel}{FEEL}}{— force-enhanced egocentric dataset}{}{
  Co-developed a large-scale dataset pairing egocentric video with calibrated
  glove-based force measurements, comprising approximately three million
  synchronized frames of natural kitchen manipulation.
}

\entry{\href{https://humanego-ai.github.io/}{HumanEgo}}{— learning from short egocentric video}{}{
  Contributed to methods for transferring human-object interaction into robot
  behavior from minutes of task video, without requiring paired robot demonstrations;
  IROS 2026 WORLDS Workshop \textbf{\textcolor{highlight}{Best Paper}}.
}

\entry{\href{https://www.far-planner.com/interactive-far-planner}{Interactive-FAR}
  \& \href{https://www.far-planner.com/air-far-planner}{Air-FAR}}{— deployable field autonomy}{}{
  Built navigation systems for ground and aerial robots operating in complex unknown
  environments. Interactive-FAR reasons about movable obstacles online; Air-FAR
  extends fast, adaptable routing to large-scale aerial navigation.
}

\entry{Microsaccade-Inspired Event Camera}{— sensing hardware and algorithms}{}{
  Co-designed an actuated event-camera mechanism and computational pipeline that
  reveals static scene structure while preserving compatibility with established
  event-based perception methods; published in \textit{Science Robotics}.
}
